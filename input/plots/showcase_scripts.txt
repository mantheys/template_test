# Beyond the official examples: what a DUNE analysis usually draws, each
# figure one plotipus command over the pickles of scripts/make_example_data.py.

# One curve per detector configuration and sample ({config}_{name}_Efficiency.pkl), with error bands
line --path examples --datafile Efficiency --configs hd_1x2x6 hd_1x2x6 vd_1x8x14 vd_1x8x14 --names marley gamma marley gamma -x Energy -y Efficiency --by_config --plot_type line --errory --errory_type bands --labelx 'True Energy (MeV)' --labely 'Efficiency (%)' --rangey 0 110 --legend_loc 'lower right'
# Binned estimator over per-event arrays: energy resolution (RMS) per configuration
line --path examples --datafile Events -x TrueEnergy -y Resolution --hue Config --iterable_mapping config_dict --iterable_color_mapping config_color --binned --estimator rms --bins 14 --plot_type errorbar --labelx 'True Energy (MeV)' --labely 'Energy Resolution (RMS)' --dune_headroom 0.3
# 2D histogram per configuration with the y = x diagonal
map --path examples --datafile Events -x TrueEnergy -y RecoEnergy --iterable Config --panel_title 'HD 1x2x6' 'VD 1x8x14' --diagonal --logz --bins 60 --labelx 'True Energy (MeV)' --labely 'Reco Energy (MeV)'
# Overlaid histograms of per-event arrays, one per configuration
hist --path examples --datafile Events -x RecoEnergy --hue Config --bins 40 --rangex 0 35 --labelx 'Reco Energy (MeV)' --labely Events --dune_label preliminary
